const express = require("express"); const cors = require("cors"); const app = express(); app.use(cors()); app.use(express.json({ limit: "10mb" })); const PORT = process.env.PORT || 3000; /* ======================================================== TELEGRAM THEME SERVER ======================================================== Mini App отправляет сюда: { name: "Pink Night", colors: { background: "#111114", header: "#18181d", accent: "#ff4fa3", outgoing: "#ff4fa3", incoming: "#292930", text: "#ffffff", link: "#ff4fa3", secondary: "#9999a3" } } ======================================================== */ function hexToInt(hex) { hex = String(hex) .replace("#", "") .trim(); if (!/^[0-9a-fA-F]{6}$/.test(hex)) { return 0; } return parseInt(hex, 16); } function makeSlug(name) { return String(name || "telegram-theme") .toLowerCase() .replace(/[^a-z0-9]+/g, "_") .replace(/^_+|_+$/g, "") .slice(0, 50) || "telegram_theme"; } /* ======================================================== ПОЛУЧИТЬ ТЕМУ ======================================================== */ app.post("/api/theme", async (req, res) => { try { const { name, colors } = req.body; if (!colors) { return res.status(400).json({ ok: false, error: "Не переданы цвета" }); } const themeName = name || "My Telegram Theme"; const slug = makeSlug(themeName); /* Telegram ThemeSettings: accent_color outbox_accent_color message_colors base_theme */ const accent = hexToInt( colors.accent ); const outgoing = hexToInt( colors.outgoing ); const incoming = hexToInt( colors.incoming ); const settings = { day: { base_theme: "day", accent_color: accent, outbox_accent_color: outgoing, message_colors: [ outgoing, incoming ] }, night: { base_theme: "night", accent_color: accent, outbox_accent_color: outgoing, message_colors: [ outgoing, incoming ] } }; /* ------------------------------------------------ ЗДЕСЬ БУДЕТ ВЫЗОВ TELEGRAM MTProto ------------------------------------------------ account.uploadTheme ↓ account.createTheme После создания Telegram возвращает slug. ------------------------------------------------ */ /* ВРЕМЕННО возвращаем данные, чтобы Mini App уже мог работать. */ const fakeTelegramSlug = slug; const link = "https://t.me/addtheme/" + encodeURIComponent( fakeTelegramSlug ); res.json({ ok: true, name: themeName, slug: fakeTelegramSlug, link: link, settings: settings }); } catch(error) { console.error(error); res.status(500).json({ ok: false, error: "Ошибка создания темы" }); } }); /* ======================================================== HEALTH CHECK ======================================================== */ app.get("/", (req, res) => { res.json({ ok: true, service: "Telegram Theme Studio" }); }); app.listen(PORT, () => { console.log( "Theme server running on port " + PORT ); });
